Context: Sarathi is an open-source Node.js/Express/TypeScript ride-sharing (driver-hire) backend (https://github.com/AkhilMM2000/sarathi-backend, commit c8766f4). It uses MongoDB/Mongoose, tsyringe dependency injection, Socket.IO, Upstash Redis, Stripe and the Google Distance Matrix API. Line numbers are shown at the left of every source line.
New requirement (Agile change CH01 - new driver status): an admin must be able to temporarily SUSPEND an approved driver (reason required) and later REINSTATE the driver. A suspended driver must not receive ride requests and must not be able to accept a booking. Today a driver status is only pending/approved/rejected and the admin can move a driver from any status to any status.

  15   constructor(
  16     @inject(TOKENS.IBOOKING_REPO) private _bookingRepo: IBookingRepository,

  18   ) {}
  19 
  20   async execute(data: AcceptBookingInput): Promise<Booking> {
  21     const { bookingId, driverId } = data;
  22 
  23     // Atomic update: only updates status to CONFIRMED if status is currently PENDING
  24     const updatedBooking = await BookingModel.findOneAndUpdate(
  25       { _id: new Types.ObjectId(bookingId), status: BookingStatus.PENDING },
  26       { status: BookingStatus.CONFIRMED, driverId: new Types.ObjectId(driverId) },
  27       { new: true }
  28     ).populate("driverId", "name profileImage mobile");
  29 
  30     // If updatedBooking is null, it was either already accepted or expired
  31     if (!updatedBooking) {
  32       throw new AuthError("This booking request is no longer available.", HTTP_STATUS_CODES.BAD_REQUEST);
  33     }
  34 
  35     const driverObj = updatedBooking.driverId as unknown as PopulatedDriver;
  36 
  37     // Notify user via Socket.IO
  38     await this._notificationService.sendBookingConfirmation(updatedBooking.userId.toString(), {
  39       bookingId: (updatedBooking._id as Types.ObjectId).toString(),
  40       status: BookingStatus.CONFIRMED,
  41       driver: driverObj ? {
  42         _id: driverObj._id.toString(),
  43         name: driverObj.name,
  44         mobile: driverObj.mobile,
  45         profileImage: driverObj.profileImage || "N/A"
  46       } : undefined
  47     });
  48 
  49     // Notify other online drivers to dismiss their popups
  50     this._notificationService.bookingAssignedNotification((updatedBooking._id as Types.ObjectId).toString(), driverId);
  51 
  52     return updatedBooking.toObject() as Booking;
  53   }
