Context: Sarathi is an open-source Node.js/Express/TypeScript ride-sharing (driver-hire) backend (https://github.com/AkhilMM2000/sarathi-backend, commit c8766f4). It uses MongoDB/Mongoose, tsyringe dependency injection, Socket.IO, Upstash Redis, Stripe and the Google Distance Matrix API. Line numbers are shown at the left of every source line.
New requirement (Agile change CH02 - ride cancellation): both the rider and the assigned driver can cancel a ride. Only PENDING or CONFIRMED rides can be cancelled; COMPLETED, CANCELLED, REJECTED or EXPIRED rides must be refused. A rider cancelling a CONFIRMED ride after 9:00 PM IST on the day before the ride pays a late cancellation fee of 10% of the estimated fare (minimum Rs 50); all other rider cancellations are free. A driver cancellation is free for the rider and the rider must be notified. Only the rider or driver of that booking may cancel it.

  11

  13     setInterval(async () => {

  15         const limitMinutes = parseInt(process.env.BOOKING_EXPIRATION_LIMIT_MINUTES || "3", 10);
  16         const expirationTime = new Date(Date.now() - limitMinutes * 60 * 1000);
  17         
  18         // Find all pending bookings created more than the limit time ago
  19         const expiredBookings = await BookingModel.find({
  20           status: BookingStatus.PENDING,
  21           createdAt: { $lt: expirationTime }
  22         });
  23 
  24         if (expiredBookings.length === 0) return;
  25 
  26         const notificationService = container.resolve<INotificationService>(TOKENS.NOTIFICATION_SERVICE);
  27 
  28         for (const booking of expiredBookings) {
  29           booking.status = BookingStatus.EXPIRED;
  30           booking.reason = `No driver accepted the request within ${limitMinutes} minutes.`;
  31           await booking.save();
  32 
  33           // 1. Notify the user
  34           notificationService.rejectBookingNotification(booking.userId.toString(), {
  35             bookingId: (booking._id as Types.ObjectId).toString(),
  36             status: BookingStatus.EXPIRED,
  37             startDate: booking.startDate,
  38             reason: booking.reason
  39           });
  40 
  41           // 2. Dismiss request popup for all online drivers
  42           notificationService.bookingAssignedNotification((booking._id as Types.ObjectId).toString());
  43         }
  44 
  45         console.log(`[CronScheduler] Expired ${expiredBookings.length} pending booking(s).`);
  46       } catch (error) {

  48       }
